import {test,expect,type Page} from '@playwright/test';
import {writeFile} from 'node:fs/promises';
import path from 'node:path';
import {registerSyntheticUser,syntheticUser} from './support/synthetic-auth';
import {setCivilDateOnly} from './support/civil-date-clock';
const dir=process.env.CF_REVIEW_EVIDENCE!,ns=process.env.CF_E2E_NAMESPACE!;
async function start(page:Page,suffix:string,clock:'playwright'|'civil'){
 if(clock==='playwright')await page.clock.setFixedTime(new Date('2027-01-15T12:00:00Z'));
 else await setCivilDateOnly(page,new Date('2027-01-15T12:00:00Z'));
 await registerSyntheticUser(page,syntheticUser('A',`${ns}-${suffix}`));
}
async function post(page:Page,url:string,data:unknown){const r=await page.request.post('/api'+url,{data});expect(r.ok(),await r.text()).toBe(true);return (await r.text())?r.json():null;}
async function json(page:Page,url:string){const r=await page.request.get('/api'+url);expect(r.ok(),await r.text()).toBe(true);return r.json();}
test.afterEach(async({page},info)=>{
 const metadata=await page.evaluate(()=>{
  const dialogs=[...document.querySelectorAll('[role="dialog"]')].map(el=>{
   const style=getComputedStyle(el),r=el.getBoundingClientRect();
   return {name:el.getAttribute('aria-label')??el.getAttribute('aria-labelledby'),connected:el.isConnected,display:style.display,visibility:style.visibility,opacity:style.opacity,rect:{x:r.x,y:r.y,width:r.width,height:r.height},html:el.outerHTML.slice(0,240),animations:el.getAnimations({subtree:true}).map(a=>({startTime:a.startTime,currentTime:a.currentTime,playState:a.playState,timing:a.effect?.getComputedTiming()}))};
  });
  return {date:Date.now(),performance:performance.now(),timeOrigin:performance.timeOrigin,timeline:document.timeline.currentTime,viewport:{width:innerWidth,height:innerHeight,dpr:devicePixelRatio},dialogs};
 }).catch(()=>({unavailable:true}));
 await writeFile(path.join(dir,`observer-${info.title.replace(/[^a-zA-Z0-9]/g,'-')}.json`),JSON.stringify({title:info.title,status:info.status,metadata},null,2));
});
for(const clock of ['playwright','civil'] as const)for(const initial of [360,1440])test(`REVIEW geometry ${clock} ${initial} to 360`,async({page})=>{
 await page.setViewportSize({width:initial,height:900});await start(page,`geometry-${clock}-${initial}`,clock);
 const investment=await post(page,'/investments',{ticker:'SANVIS',name:'Avaliação desconhecida',type:'STOCK',openingQuantity:'10',openingCost:'200.00'});
 await page.goto('/investimentos');
 const reason=page.getByText('Custo ou avaliação indisponível',{exact:true});await expect(reason).toBeVisible();
 const read=()=>reason.evaluate(el=>{
  const r=el.getBoundingClientRect(),ancestors=[];
  for(let a:Element|null=el;a;a=a.parentElement){const s=getComputedStyle(a),r=a.getBoundingClientRect();ancestors.push({tag:a.tagName,opacity:s.opacity,display:s.display,visibility:s.visibility,transform:s.transform,rect:{x:r.x,y:r.y,width:r.width,height:r.height}});}
  return {connected:el.isConnected,text:el.textContent,date:Date.now(),performance:performance.now(),timeOrigin:performance.timeOrigin,timeline:document.timeline.currentTime,rect:{x:r.x,y:r.y,width:r.width,height:r.height},hit:document.elementFromPoint(r.x+r.width/2,r.y+r.height/2)?.textContent?.slice(0,100),ancestors,animations:document.getAnimations().map(a=>({startTime:a.startTime,currentTime:a.currentTime,playState:a.playState}))};
 });
 const before=await read();await page.setViewportSize({width:360,height:900});await reason.scrollIntoViewIfNeeded();
 const box=await reason.boundingBox(),atMeasure=await read();
 const actual=(await json(page,'/investments')).find((i:{id:string})=>i.id===investment.id);
 await writeFile(path.join(dir,`geometry-${clock}-${initial}.json`),JSON.stringify({investmentId:investment.id,actual,before,box,atMeasure},null,2));
 await page.screenshot({path:path.join(dir,`geometry-${clock}-${initial}-live.png`),animations:'allow'});
 await page.screenshot({path:path.join(dir,`geometry-${clock}-${initial}.png`),animations:'disabled'});
 await writeFile(path.join(dir,`geometry-${clock}-${initial}-after-capture.json`),JSON.stringify(await read(),null,2));
 expect(actual.ticker).toBe('SANVIS');expect(actual.quantity).toBe(10);expect(actual.totalCost).toBe(200);expect(actual.valuationAmount).toBeNull();
 expect(box).not.toBeNull();expect(box!.x).toBeGreaterThanOrEqual(0);expect(box!.x+box!.width).toBeLessThanOrEqual(360);
});
