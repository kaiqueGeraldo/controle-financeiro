package com.controlefinanceiro.api.domain.onboarding;

import com.controlefinanceiro.api.domain.investment.*;
import com.controlefinanceiro.api.domain.investment.dto.*;

import com.controlefinanceiro.api.service.exception.BusinessRuleException;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.params.ParameterizedTest;
import org.junit.jupiter.params.provider.ValueSource;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.test.context.bean.override.mockito.MockitoSpyBean;
import org.springframework.test.util.AopTestUtils;
import org.springframework.transaction.support.TransactionTemplate;
import java.math.BigDecimal;
import java.nio.file.*;
import java.time.LocalDate;
import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.*;
import static org.assertj.core.api.Assertions.*;

/** Review-only observer: call the real predicate, then pause its first result.
 * Independent connections execute actual services. No financial answer is stubbed. */
class Cf033GateTransitionReviewTest extends OnboardingApplicationSupport {
    @Autowired InvestmentService investments;
    @MockitoSpyBean FinancialInventoryGuard observedGuard;
    static final LocalDate DAY=LocalDate.of(2026,10,19);
    UUID goal() {
        UUID id=UUID.randomUUID();
        sql.update("insert into goals(id,title,type,target_value,current_value,deadline,user_id,allocation_reconciled,version) values(?,'Review','MONETARY',2000,0,'2027-10-23',?,true,0)",id,user.getId());
        return id;
    }
    UUID position(UUID goal) {
        return investments.create(new CreateInvestmentDTO("RV033","Synthetic",InvestType.STOCK,goal),user).id();
    }
    InvestmentOperationDTO buy(UUID position,UUID source) {
        return new InvestmentOperationDTO(position,InvestTransType.BUY,new BigDecimal("2"),new BigDecimal("100.00"),BigDecimal.ZERO,DAY,source);
    }
    BigDecimal balance(UUID source) {return sql.queryForObject("select balance from accounts where id=?",BigDecimal.class,source);}
    void outcome(UUID source,UUID investment,boolean committed) {
        assertThat(balance(source)).isEqualByComparingTo(committed?"800.00":"1000.00");
        assertThat(sql.queryForObject("select quantity from investments where id=?",BigDecimal.class,investment)).isEqualByComparingTo(committed?"2":"0");
        assertThat(sql.queryForObject("select coalesce(sum(amount),0) from goal_allocations where account_id=?",BigDecimal.class,source)).isEqualByComparingTo("0.00");
    }
    String attempt(UUID investment,UUID source,AtomicInteger pid,CountDownLatch started) {
        try {
            new TransactionTemplate(txManager).executeWithoutResult(s->{
                pid.set(sql.queryForObject("select pg_backend_pid()",Integer.class));started.countDown();
                investments.performOperation(buy(investment,source),user);
            });
            return "COMMITTED";
        } catch(BusinessRuleException e) {
            assertThat(e.getMessage()).contains("Capacidade da fonte não comprovada");
            return "REJECTED_UNKNOWN";
        }
    }
    void reached(Future<?> task,CountDownLatch latch) throws Exception {
        long end=System.nanoTime()+TimeUnit.SECONDS.toNanos(10);
        while(latch.getCount()>0 && !task.isDone() && System.nanoTime()<end)Thread.onSpinWait();
        if(task.isDone())task.get(15,TimeUnit.SECONDS);
        await(latch);
    }
    Map<String,Object> waiting(Future<?> task,int pid) {
        long end=System.nanoTime()+TimeUnit.SECONDS.toNanos(10);
        while(!task.isDone() && !blocked(pid) && System.nanoTime()<end)Thread.onSpinWait();
        assertThat(blocked(pid)).as("real PostgreSQL blocker before holder release; completed=%s",task.isDone()).isTrue();
        var row=sql.queryForMap("select pid,wait_event,query,pg_blocking_pids(pid)::text as blockers from pg_stat_activity where pid=?",pid);
        assertThat(row.get("wait_event")).isEqualTo("advisory");return row;
    }
    boolean blocked(int pid) {return Boolean.TRUE.equals(sql.queryForObject("select cardinality(pg_blocking_pids(?))>0",Boolean.class,pid));}
    void observeFirstPredicate(AtomicReference<Thread> thread,CountDownLatch classified,CountDownLatch release,boolean expected) {
        FinancialInventoryGuard spy=AopTestUtils.getUltimateTargetObject(observedGuard);
        var first=new AtomicBoolean(true);
        doAnswer(invocation->{
            boolean actual=(Boolean)invocation.callRealMethod();
            if(Thread.currentThread()==thread.get() && first.compareAndSet(true,false)) {
                assertThat(actual).as("real dependency predicate before choosing inventory mode").isEqualTo(expected);
                classified.countDown();await(release);
            }
            return actual;
        }).when(spy).hasInventoryDependencies(user.getId());
    }
    void evidence(String name,Map<String,Object> facts) throws Exception {
        String out=System.getenv("CF032_EVIDENCE");
        var record=new LinkedHashMap<String,Object>(facts);
        record.put("database",sql.queryForMap("select current_database() as database,current_schema() as schema,current_setting('transaction_isolation') as isolation,current_setting('TimeZone') as timezone,inet_server_port() as port,version() as version"));
        record.put("migrations",sql.queryForList("select version,success from flyway_schema_history where version is not null order by installed_rank"));
        if(out!=null)Files.writeString(Path.of(out,name+".json"),OnboardingJson.JSON.writeValueAsString(record),StandardOpenOption.CREATE_NEW);
    }

    @Test void dependencyCannotCommitInGapAfterLocalClassification() throws Exception {
        UUID source=account("1000.00"),investment=position(null),g=goal();
        var thread=new AtomicReference<Thread>();var classified=new CountDownLatch(1);var release=new CountDownLatch(1);
        observeFirstPredicate(thread,classified,release,false);
        var creatorStarted=new CountDownLatch(1);var pid=new AtomicInteger();Map<String,Object> facts=new LinkedHashMap<>();
        try(var pool=Executors.newVirtualThreadPerTaskExecutor()) {
            var buyer=pool.submit(()->{thread.set(Thread.currentThread());investments.performOperation(buy(investment,source),user);});
            try {
                reached(buyer,classified);
                var creator=pool.submit(()->new TransactionTemplate(txManager).execute(s->{
                    pid.set(sql.queryForObject("select pg_backend_pid()",Integer.class));creatorStarted.countDown();return position(g);
                }));
                try {
                    await(creatorStarted);facts.put("creatorBlockedAfterFalseBeforeInventoryLock",waiting(creator,pid.get()));
                    assertThat(sql.queryForObject("select count(*) from investments where user_id=? and goal_id is not null",Integer.class,user.getId())).isZero();
                } finally {release.countDown();}
                buyer.get(15,TimeUnit.SECONDS);assertThat(creator.get(15,TimeUnit.SECONDS)).isNotNull();
            } finally {release.countDown();}
        }
        assertThatThrownBy(()->investments.performOperation(buy(investment,source),user)).isInstanceOf(BusinessRuleException.class);
        outcome(source,investment,true);facts.put("balance",balance(source));facts.put("nextBuyRejected",true);evidence("gap-creation",facts);
    }

    @Test void dependencyRemovedAfterGlobalClassificationBeforeInventoryLock() throws Exception {
        UUID source=account("1000.00"),investment=position(null),legacy=position(goal());
        var thread=new AtomicReference<Thread>();var classified=new CountDownLatch(1);var release=new CountDownLatch(1);
        observeFirstPredicate(thread,classified,release,true);
        try(var pool=Executors.newVirtualThreadPerTaskExecutor()) {
            var buyer=pool.submit(()->{thread.set(Thread.currentThread());investments.performOperation(buy(investment,source),user);});
            try {
                reached(buyer,classified);
                pool.submit(()->investments.deleteInvestment(legacy,user)).get(4,TimeUnit.SECONDS);
                assertThat(sql.queryForObject("select count(*) from investments where id=?",Integer.class,legacy)).isZero();
                assertThat(buyer.isDone()).isFalse();
            } finally {release.countDown();}
            buyer.get(15,TimeUnit.SECONDS);
        }
        outcome(source,investment,true);
        evidence("gap-removal",Map.of("removedBeforeBuyerInventoryLock",true,"recheckedAtUse",true,"balance",balance(source)));
    }

    @ParameterizedTest @ValueSource(booleans={false,true})
    void pendingCreationIsRecheckedAfterCommitOrRollback(boolean rollback) throws Exception {
        UUID source=account("1000.00"),investment=position(null),g=goal();
        var held=new CountDownLatch(1);var release=new CountDownLatch(1);var started=new CountDownLatch(1);var pid=new AtomicInteger();
        Map<String,Object> facts=new LinkedHashMap<>();
        try(var pool=Executors.newVirtualThreadPerTaskExecutor()) {
            var creator=pool.submit(()->new TransactionTemplate(txManager).executeWithoutResult(s->{position(g);held.countDown();await(release);if(rollback)s.setRollbackOnly();}));
            try {
                reached(creator,held);
                var buyer=pool.submit(()->attempt(investment,source,pid,started));
                try {await(started);facts.put("blocked",waiting(buyer,pid.get()));}finally{release.countDown();}
                creator.get(15,TimeUnit.SECONDS);
                assertThat(buyer.get(15,TimeUnit.SECONDS)).isEqualTo(rollback?"COMMITTED":"REJECTED_UNKNOWN");
            }finally{release.countDown();}
        }
        assertThat(sql.queryForObject("select count(*) from investments where user_id=? and goal_id is not null",Integer.class,user.getId())).isEqualTo(rollback?0:1);
        outcome(source,investment,rollback);facts.put("rollback",rollback);facts.put("balance",balance(source));evidence("creation-"+(rollback?"rollback":"commit"),facts);
    }

    @ParameterizedTest @ValueSource(booleans={false,true})
    void pendingRemovalIsRecheckedAfterInventoryWait(boolean rollback) throws Exception {
        UUID source=account("1000.00"),investment=position(null),legacy=position(goal());
        var held=new CountDownLatch(1);var release=new CountDownLatch(1);var started=new CountDownLatch(1);var pid=new AtomicInteger();
        Map<String,Object> facts=new LinkedHashMap<>();
        try(var pool=Executors.newVirtualThreadPerTaskExecutor()) {
            var remover=pool.submit(()->new TransactionTemplate(txManager).executeWithoutResult(s->{investments.deleteInvestment(legacy,user);held.countDown();await(release);if(rollback)s.setRollbackOnly();}));
            try {
                reached(remover,held);
                var buyer=pool.submit(()->attempt(investment,source,pid,started));
                try {await(started);facts.put("blocked",waiting(buyer,pid.get()));}finally{release.countDown();}
                remover.get(15,TimeUnit.SECONDS);
                assertThat(buyer.get(15,TimeUnit.SECONDS)).isEqualTo(rollback?"REJECTED_UNKNOWN":"COMMITTED");
            }finally{release.countDown();}
        }
        assertThat(sql.queryForObject("select count(*) from investments where id=?",Integer.class,legacy)).isEqualTo(rollback?1:0);
        outcome(source,investment,!rollback);facts.put("rollback",rollback);facts.put("balance",balance(source));evidence("removal-"+(rollback?"rollback":"commit"),facts);
    }
}
